# Python While Loops

>>> i = 1
>>> while i < 6:
...     print(i)
...     i += 1
...
1
2
3
4
5

>>> i = 1
>>> while i < 6:
...     print(i)
...     if i == 3:
...         break
...     i += 1
...
1
2
3

>>> i = 0
>>> while i < 6:
...     i += 1
...     if i == 3:
...         continue
...     print(i)
...
1
2
4
5
6

# Python For Loops

>>> fruits = ["apple", "banana", "cherry"]
>>> for x in fruits:
...     print(x)
...
apple
banana
cherry

>>> for x in range(6):
...     print(x)
...
0
1
2
3
4
5

>>> for x in range(6):
...     if x == 3:
...         break
...     print(x)
...
0
1
2

>>> for x in fruits:
...     if x == "banana":
...         continue
...     print(x)
...
apple
cherry

>>> for x in [0, 1, 2]:
...     for y in ["a", "b"]:
...         print(x, y)
...
0 a
0 b
1 a
1 b
2 a
2 b
