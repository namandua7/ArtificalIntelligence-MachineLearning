# Python Functions

>>> def my_function():
...     print("Hello World")
...
>>> my_function()
Hello World

>>> def my_function(fname):
...     print(fname + " Dua")
...
>>> my_function("Naman")
Naman Dua

>>> def my_function(country="India"):
...     print("I am from " + country)
...
>>> my_function("Vietnam")
I am from Vietnam
>>> my_function()
I am from India

>>> def my_function(x):
...     return 5 * x
...
>>> print(my_function(3))
15
>>> print(my_function(5))
25

>>> def my_function(x):
...     if x > 0:
...         return True
...     else:
...         return False
...
>>> print(my_function(2))
True
