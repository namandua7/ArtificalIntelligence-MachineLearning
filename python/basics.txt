# Python Strings

>>> a = "Hello, World!"
>>> print(a[1])
e
>>> print(a[2:5])
llo
>>> print(len(a))
13
>>> print(a.upper())
HELLO, WORLD!
>>> print(a.lower())
hello, world!
>>> print(a.strip())

>>> print(a.replace("H", "J"))
